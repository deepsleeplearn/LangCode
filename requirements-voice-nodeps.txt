# Installed with `pip install --no-deps -c constraints.txt -r <this file>`.

# Both packages declare requirements that cannot be satisfied together:
#   qwen-asr==0.0.6   -> transformers==4.57.6 (+ gradio/flask/accelerate/... demo tail)
#   mlx-audio==0.4.3  -> transformers>=5.5.0, huggingface_hub>=1.0
#   mlx-lm==0.31.3    -> transformers>=5.0.0
#
# transformers 4.57.6 + mlx-audio 0.4.3 is the combination that actually works at
# runtime, so we assert it instead of letting pip backtrack looking for one.
# Their real runtime imports are listed in the `voice` extra of pyproject.toml.
qwen-asr==0.0.6
mlx-audio==0.4.3
mlx-lm==0.31.3
